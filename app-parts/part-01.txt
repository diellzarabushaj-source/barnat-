(async () => {
const CANONICAL_FIELDS = [
  'Nr rendor','PDID','ProtocolNo','Emri tregtar','Substanca aktive','ATC Code',
  'Klasa / Çka është','Përdorimi (fjalë kyçe)','Fortësia','Forma farmaceutike',

const REGISTRY_WORKER_URL = './registry-parser-worker-v2.js';
const REGISTRY_WORKER_TIMEOUT_MS = 30000;
const BASE64_CHUNK = 128 * 1024;
const NORMALIZE_BATCH = 120;

function yieldToBrowser(){
  return new Promise(resolve => {
    if(typeof requestAnimationFrame === 'function') requestAnimationFrame(() => resolve());
    else setTimeout(resolve, 0);
  });
}

function normalizeFieldToken(value){
  return String(value ?? '')
    .normalize('NFD').replace(/[\u0300-\u036f]/g, '')
    .toLowerCase()
    .replace(/&[a-z]+;/g, '')
    .replace(/[^a-z0-9]+/g, '');
}

const FIELD_ALIASES = {
  'Nr rendor':['nrrendor','nr','number','index','rendor'],
  'PDID':['pdid','productid','drugid','id'],
  'ProtocolNo':['protocolno','protocol','protokolli','protokollno'],
  'Emri tregtar':['emritregtar','tradename','brandname','emri','name'],
  'Substanca aktive':['substancaaktive','activesubstance','activeingredient','genericname','substanca'],

  'Klasa / Çka është':['klasackaeshte','klasa','class','drugclass','ckaeshte'],

  'Fortësia':['fortesia','strength','dosestrength'],
  'Forma farmaceutike':['formafarmaceutike','forma','dosageform','pharmaceuticalform'],
  'Madhësia e paketimit':['madhesiaepaketimit','paketimi','packsize','packagesize'],
  'Si të shënohet në recetë':['siteshenohetnerecete','sitetshenohetnerecete','prescriptionnotation','prescriptionline','receta'],
  'Bartësi i Autorizim Marketingut':['bartesiiautorizimmarketingut','bartesiiautorizimit','marketingauthorisationholder','mah'],

  'Çmimi me marzhë':['cmimimemarzhe','marginprice'],

  'Çmimi me pakicë':['cmimimepakice','retailprice','price'],
  'Afati i vlefshmërisë':['afatiivlefshmerise','validity','validuntil','expiry']
};

const FIELD_LOOKUP = {};
CANONICAL_FIELDS.forEach(field => {
  FIELD_LOOKUP[normalizeFieldToken(field)] = field;
  (FIELD_ALIASES[field] || []).forEach(alias => { FIELD_LOOKUP[normalizeFieldToken(alias)] = field; });
});

function unwrapDrugRows(value){
  let current = value;
  for(let depth = 0; depth < 5; depth++){
    if(Array.isArray(current)) return current;
    if(typeof current === 'string'){
      try { current = JSON.parse(current); continue; } catch { return []; }
    }
    if(current && typeof current === 'object'){
      const preferred = ['data','rows','records','items','drugs','barnat','Sheet1','sheet1'];
      const preferredKey = preferred.find(key => Array.isArray(current[key]) || typeof current[key] === 'string');
      if(preferredKey){ current = current[preferredKey]; continue; }
      const nested = Object.values(current).find(Array.isArray);
      if(nested){ current = nested; continue; }
    }
    break;
  }
  return [];
}

function normalizeDrugRow(row, index){
  let sourceRow = row;
  if(typeof sourceRow === 'string'){
    try { sourceRow = JSON.parse(sourceRow); } catch { sourceRow = { 'Emri tregtar':sourceRow }; }
  }
  const result = Object.fromEntries(CANONICAL_FIELDS.map(field => [field, '']));
  if(Array.isArray(sourceRow)){
    CANONICAL_FIELDS.forEach((field, position) => { result[field] = sourceRow[position] ?? ''; });
  } else if(sourceRow && typeof sourceRow === 'object'){
    const source = sourceRow.data && typeof sourceRow.data === 'object' && !Array.isArray(sourceRow.data) ? sourceRow.data : sourceRow;
    Object.entries(source).forEach(([key, value]) => {
      const canonical = FIELD_LOOKUP[normalizeFieldToken(key)];
      if(canonical) result[canonical] = value ?? '';
    });
  }
  if(result['Nr rendor'] === '' || result['Nr rendor'] == null) result['Nr rendor'] = index + 1;
  return result;
}

async function normalizeDrugRows(value){
  const rows = unwrapDrugRows(value);
  const output = new Array(rows.length);
  for(let start = 0; start < rows.length; start += NORMALIZE_BATCH){
    const end = Math.min(rows.length, start + NORMALIZE_BATCH);
    for(let index = start; index < end; index++) output[index] = normalizeDrugRow(rows[index], index);
    if(end < rows.length) await yieldToBrowser();
  }
  return output;
}

function updateWorkerProgress(stage){
  const badge = document.getElementById('countBadge');
  if(!badge || !/ngarkohet|përgatitet|duke/i.test(badge.textContent || '')) return;
  const labels = { decode:'Po lexohet regjistri…', decompress:'Po hapet databaza…', normalize:'Po përgatiten barnat…' };
  if(labels[stage]) badge.textContent = labels[stage];
}

function parseRegistryInWorker(parts){
  return new Promise((resolve, reject) => {
    if(typeof Worker !== 'function') return reject(new Error('Web Worker nuk mbështetet.'));
    let settled = false;
    const worker = new Worker(REGISTRY_WORKER_URL);
    const finish = (callback, value) => {
      if(settled) return;
      settled = true;
      clearTimeout(timeout);
      worker.terminate();
      callback(value);
    };
    const timeout = setTimeout(() => finish(reject, new Error('Përpunimi i regjistrit zgjati tepër.')), REGISTRY_WORKER_TIMEOUT_MS);
    worker.addEventListener('message', event => {
      if(event.data?.type === 'REGISTRY_PROGRESS'){
        updateWorkerProgress(event.data.stage);
        return;
      }
      if(event.data?.type !== 'REGISTRY_PARSED') return;
      if(event.data.ok && Array.isArray(event.data.data?.rows)) finish(resolve, event.data.data);
      else finish(reject, new Error(event.data.error || 'Worker-i nuk e lexoi regjistrin.'));
    });
    worker.addEventListener('error', () => finish(reject, new Error('Worker-i i regjistrit dështoi.')));
    worker.postMessage({ type:'PARSE_REGISTRY', parts });
  });
}

async function parseRegistryCooperatively(parts){
  const encoded = Array.isArray(parts) ? parts.join('') : '';
  if(!encoded) return [];
  const chunks = [];
  let total = 0;
  let offset = 0;
  while(offset < encoded.length){
    let end = Math.min(encoded.length, offset + BASE64_CHUNK);
    if(end < encoded.length) end -= (end - offset) % 4;
    if(end <= offset) end = Math.min(encoded.length, offset + 4);
    const binary = atob(encoded.slice(offset, end));
    const bytes = new Uint8Array(binary.length);
    for(let index = 0; index < binary.length; index++) bytes[index] = binary.charCodeAt(index);
    chunks.push(bytes);
    total += bytes.length;
    offset = end;
    await yieldToBrowser();
  }
  const compressed = new Uint8Array(total);
  let cursor = 0;
  for(const chunk of chunks){ compressed.set(chunk, cursor); cursor += chunk.length; }
  if(typeof DecompressionStream !== 'function') throw new Error('Browser-i nuk e mbështet dekompresimin e regjistrit.');
  const stream = new Blob([compressed]).stream().pipeThrough(new DecompressionStream('gzip'));
  return JSON.parse(await new Response(stream).text());
}

async function ensureRegistryQuality(){
  if(window.MedIndexRegistryQuality) return window.MedIndexRegistryQuality;
  await new Promise(resolve => {
    let settled = false;
    const finish = () => {
      if(settled) return;
      settled = true;
      clearTimeout(timeout);
      resolve();
    };
    const timeout = setTimeout(finish, 2600);
    const existing = document.querySelector('script[data-registry-quality]');
    if(existing){
      if(window.MedIndexRegistryQuality) return finish();
      existing.addEventListener('load', finish, { once:true });
      existing.addEventListener('error', finish, { once:true });
      return;
    }
    const script = document.createElement('script');
    script.src = './data/registry-quality.js?v=20260723-2';
    script.dataset.registryQuality = '1';
    script.async = true;
    script.onload = finish;
    script.onerror = finish;
    document.head.appendChild(script);
  });
  return window.MedIndexRegistryQuality || null;
}

function emergencyRegistryCorrection(rows){
  let corrected = 0;
  const output = rows.map(row => {
    if(String(row.ProtocolNo ?? '').trim() !== 'PD1339/051225' || String(row.PDID ?? '').trim() !== '42') return row;
    corrected += 1;
    return {
      ...row,
      'Substanca aktive':'Metamizole sodium',
      __qualityStatus:'corrected',
      __qualityMessage:'REG-2026-001: substanca u korrigjua në Metamizole sodium.',
      __qualitySourceUrl:'https://lekovi.zdravstvo.gov.mk/drugsregister/detailview/51155',
      __qualityVersion:'emergency-performance-v1'
    };
  });
  return { rows:output, summary:{ total:output.length, corrected, blocked:0, warning:0, verified:Math.max(0, output.length - corrected) }, version:'emergency-performance-v1', corrections:[] };
}

async function buildRegistryDataset(){
  const parts = Array.isArray(window.DRUG_DATA_PARTS) ? window.DRUG_DATA_PARTS : [];
  if(parts.length){
    try { return await parseRegistryInWorker(parts); }
    catch(error){ console.warn('Worker-i dështoi; po përdoret fallback-i gradual:', error); }
  }
  const parsed = parts.length
    ? await parseRegistryCooperatively(parts)
    : JSON.parse(document.getElementById('drug-data')?.textContent || '[]');
  const normalizedRows = await normalizeDrugRows(parsed);
  await yieldToBrowser();
  const registryQuality = await ensureRegistryQuality();
  const qualityResult = registryQuality?.applyRows?.(normalizedRows) || emergencyRegistryCorrection(normalizedRows);
  const { rows, ...quality } = qualityResult;
  return { rows, quality };
}

let registryDataset;
try {
  registryDataset = await buildRegistryDataset();
} catch(error) {
  console.error('Të dhënat e barnave nuk u lexuan:', error);
  registryDataset = { rows:[], quality:{ version:'error', summary:{ total:0, corrected:0, blocked:0, warning:0, verified:0 }, corrections:[] } };
}

const RAW = Array.isArray(registryDataset.rows) ? registryDataset.rows : [];
window.MEDINDEX_REGISTRY_ROWS = RAW;
window.MEDINDEX_REGISTRY_QUALITY = { ...(registryDataset.quality || {}), rows:RAW };
window.dispatchEvent(new CustomEvent('medindex:registry-data-ready', { detail:{ rows:RAW, quality:window.MEDINDEX_REGISTRY_QUALITY } }));
if(!RAW.length) console.error('Nuk u gjet asnjë rresht i vlefshëm në regjistër.');
