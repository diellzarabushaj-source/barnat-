const REGISTRY_DEFAULT_PAGE_SIZE = 50;
const REGISTRY_ALLOWED_PAGE_SIZES = new Set([50, 100, 250, 500, 4006]);
let registryAtcRowsCacheCode = '';
let registryAtcRowsCache = RAW;

function sanitizeRegistryPageSize(value){
  const numeric = Number(value);
  return REGISTRY_ALLOWED_PAGE_SIZES.has(numeric) ? numeric : REGISTRY_DEFAULT_PAGE_SIZE;
}

// `atc-shared.js` is intent-deferred below the desktop breakpoint, so the ATC
// helper is routinely still absent when this runtime boots on a tablet. Reading
// the URL directly as a fallback keeps a shared `?atc=…&q=…` link working there
// instead of silently dropping the filter — the phone-lite runtime already
// normalizes the same way.
function normalizeRegistryAtcCode(value){
  const resolved = window.MedIndexATC?.resolveCategoryCode?.(value);
  if(resolved) return resolved;
  const compact = String(value ?? '').trim().toUpperCase().replace(/\s+/g, '');
  return /^[A-Z](?:\d{2})?$/.test(compact) ? compact : '';
}

function readRegistryUrlState(){
  const params = new URLSearchParams(window.location.search);
  const reader = window.MedIndexATC?.readRegistryUrlState;
  const value = typeof reader === 'function' ? reader(window.location.href) : {};
  const page = Number(value.page ?? params.get('page'));
  return {
    atc:normalizeRegistryAtcCode(value.atc ?? params.get('atc')),
    query:String(value.query ?? params.get('q') ?? ''),
    page:Number.isInteger(page) && page > 0 ? page : 1,
    pageSize:sanitizeRegistryPageSize(value.pageSize ?? params.get('pageSize')),
  };
}

function resetRegistryFilterCaches(){
  filteredCacheKey = '';
  filteredCacheRows = RAW;
  sortedCacheKey = '';
  sortedCacheInput = null;
  sortedCacheRows = RAW;
  registryAtcRowsCacheCode = '';
  registryAtcRowsCache = RAW;
}

function applyRegistryUrlStateFromLocation(){
  const next = readRegistryUrlState();
  state.activeAtc = next.atc;
  state.search = next.query;
  state.page = next.page;
  state.pageSize = next.pageSize;
  resetRegistryFilterCaches();

  const searchInput = document.getElementById('search');
  if(searchInput && searchInput.value !== state.search) searchInput.value = state.search;

  const pageSizeSelect = document.getElementById('pageSize');
  if(pageSizeSelect){
    const available = [...pageSizeSelect.options].some(option => Number(option.value) === state.pageSize);
    if(available) pageSizeSelect.value = String(state.pageSize);
    else {
      state.pageSize = REGISTRY_DEFAULT_PAGE_SIZE;
      pageSizeSelect.value = String(REGISTRY_DEFAULT_PAGE_SIZE);
    }
  }
}

function syncRegistryUrlState(options = {}){
  const builder = window.MedIndexATC?.registryUrlFromState;
  if(typeof builder !== 'function' || !window.history) return;
  const next = builder(window.location.href, {
    atc:state.activeAtc,
    query:state.search,
    page:state.page,
    pageSize:state.pageSize,
  });
  const current = `${window.location.pathname}${window.location.search}${window.location.hash}`;
  if(next === current) return;
  const method = options.push === true ? 'pushState' : 'replaceState';
  window.history[method]({ medindexRegistry:true }, '', next);
}

// Prefix match over the same normalization the shared helper uses, for the
// window before `atc-shared.js` arrives. Without it the registry would announce
// an active ATC category while still listing every row — a filter claimed and
// not applied, which is the one outcome a clinician must never be shown.
function matchesRegistryAtcCategory(value, categoryValue){
  const code = String(value ?? '').trim().toUpperCase().replace(/\s+/g, '');
  const category = String(categoryValue ?? '').trim().toUpperCase().replace(/\s+/g, '');
  return Boolean(code && category && code.startsWith(category));
}

function getRegistryAtcRows(){
  const activeAtc = String(state.activeAtc || '');
  if(!activeAtc) return RAW;
  if(activeAtc === registryAtcRowsCacheCode) return registryAtcRowsCache;
  const matches = window.MedIndexATC?.matchesCategory || matchesRegistryAtcCategory;
  registryAtcRowsCacheCode = activeAtc;
  registryAtcRowsCache = RAW.filter(row => matches(row['ATC Code'], activeAtc));
  return registryAtcRowsCache;
}

currentFilterKey = function currentFilterKeyWithAtc(){
  return [
    String(state.activeAtc || ''),
    normalizeSearchText(state.search),
    state.status,
    state.formType || '',
    state.formValue || '',
  ].join('|');
};

getFiltered = function getFilteredWithAtc(){
  const key = currentFilterKey();
  if(key === filteredCacheKey) return filteredCacheRows;

  let rows = getRegistryAtcRows();
  const q = normalizeSearchText(state.search);
  if(q){
    const terms = q.split(/\s+/).filter(Boolean);
    rows = rows.filter(row => rowMatchesSearch(row, terms));
  }
  if(state.status){
    rows = rows.filter(row => String(row['Statusi'] ?? '').trim() === state.status);
  }
  if(state.formType === 'form'){
    rows = rows.filter(row => String(row['Forma farmaceutike'] ?? '').trim() === state.formValue);
  } else if(state.formType === 'category'){
    rows = rows.filter(row => categoryOf(row['Forma farmaceutike']) === state.formValue);
  }

  filteredCacheKey = key;
  filteredCacheRows = rows;
  return rows;
};

const renderRegistryBase = render;
render = function renderRegistryWithAtcState(){
  renderRegistryBase();

  const categoryRows = getRegistryAtcRows();
  const filteredRows = getFiltered();
  const activeAtc = String(state.activeAtc || '');
  const label = activeAtc
    ? (window.MedIndexATC?.getCategoryLabel?.(activeAtc) || `Kategoria ATC ${activeAtc}`)
    : '';
  const detail = {
    activeAtc,
    label,
    categoryTotal:categoryRows.length,
    filteredTotal:filteredRows.length,
    query:String(state.search || ''),
    page:state.page,
    pageSize:state.pageSize,
  };

  window.MEDINDEX_REGISTRY_ACTIVE_ATC = activeAtc;
  window.MEDINDEX_REGISTRY_ATC_STATE = detail;

  if(activeAtc){
    const quality = window.MEDINDEX_REGISTRY_QUALITY?.summary || {};
    const qualityNote = quality.blocked ? ` · ${quality.blocked} të bllokuara për verifikim` : '';
    const badge = document.getElementById('countBadge');
    if(badge) badge.textContent = `${filteredRows.length} / ${categoryRows.length} barna · ${activeAtc}${qualityNote}`;

    if(filteredRows.length === 0){
      const empty = document.querySelector('#tbody .empty-state');
      if(empty){
        empty.textContent = state.search
          ? `Nuk u gjet “${state.search}” brenda ${label}.`
          : `Nuk u gjet asnjë bar në ${label}.`;
      }
    }
  }

  window.dispatchEvent(new CustomEvent('medindex:registry-atc-state', { detail }));
};

}

function currentPageRows(){
  const filtered = sortRows(getFiltered());
  const start = (state.page - 1) * state.pageSize;
  return filtered.slice(start, start + state.pageSize);
}

function isPrescriptionSelectable(row){
  return row && String(row.__qualityStatus || '').trim() !== 'blocked';
}

function toggleCurrentPageSelection(checked){
  currentPageRows().forEach(r => {
    const key = drugKey(r);
    if(!isPrescriptionSelectable(r)){
      selectedDrugKeys.delete(key);
      return;
    }
    if(checked) selectedDrugKeys.add(key); else selectedDrugKeys.delete(key);
  });
  syncDraftFromSelection();
  render();
}

function selectedRows(){
  return RAW.filter(r => isPrescriptionSelectable(r) && selectedDrugKeys.has(drugKey(r)));
}

function itemFromDrug(r){
  if(!isPrescriptionSelectable(r)) return null;
  return {
    drugKey: drugKey(r),
    key: drugKey(r),
    tradeName: String(r['Emri tregtar'] || ''),
    substance: String(r['Substanca aktive'] || ''),
    strength: String(r['Fortësia'] || ''),
    form: String(r['Forma farmaceutike'] || ''),
    atc: String(r['ATC Code'] || ''),
    pdid: String(r.PDID || ''),
    qualityStatus: String(r.__qualityStatus || 'verified')
  };
}

  const items = selectedRows().map(itemFromDrug).filter(Boolean);

}
